function startDrive() { if (!navigator.geolocation) { status("GPS not supported."); return; } followMode = true; lastGps = null; lastHeading = map.getBearing() || 0; setMenu(false); document .getElementById("followBtn") .classList.add("active"); document .getElementById("followBtn") .textContent = "🚗 Stop Drive"; document .getElementById("driveInfo") .style.display = "block"; status("Drive mode starting..."); followWatchId = navigator.geolocation.watchPosition( position => { const lat = position.coords.latitude; const lng = position.coords.longitude; const accuracy = position.coords.accuracy; const speed = position.coords.speed; /* ============================ FIND DIRECTION OF TRAVEL ============================ */ let newHeading = position.coords.heading; /* Phone GPS sometimes does not provide heading. If that happens, calculate direction from movement. */ if ( newHeading == null || Number.isNaN(newHeading) ) { if (lastGps) { const moved = distanceMetres( lastGps, { lat, lng } ); /* Only update direction if vehicle actually moved. This prevents GPS jumping. */ if (moved >= 3) { newHeading = calculateBearing( lastGps, { lat, lng } ); } } } /* ============================ SMOOTH ROTATION ============================ */ if ( newHeading != null && !Number.isNaN(newHeading) ) { /* Find shortest rotation direction so map doesn't spin 350 degrees when it only needs 10 degrees. */ let difference = ((newHeading - lastHeading + 540) % 360) - 180; /* Smooth the heading. Higher number = faster turn. Lower number = smoother. */ const smoothing = 0.35; lastHeading = ( lastHeading + difference * smoothing + 360 ) % 360; } /* ============================ UPDATE ARROW ============================ */ updateGpsMarker( lng, lat, 0 ); /* IMPORTANT: Arrow rotation is 0 because MAP itself rotates. This keeps arrow pointing straight toward top of screen, like navigation apps. */ /* ============================ MAP FOLLOWS CAR ============================ */ let drivingZoom = 17.5; /* Zoom out slightly at higher speed so you can see more road ahead. */ if (speed != null) { const kmh = speed * 3.6; if (kmh > 80) { drivingZoom = 15.8; } else if (kmh > 50) { drivingZoom = 16.3; } else if (kmh > 25) { drivingZoom = 16.8; } } map.easeTo({ center: [ lng, lat ], /* Rotate whole map to driving direction. */ bearing: lastHeading, zoom: drivingZoom, /* Slight angle gives more navigation-style appearance. Change to 0 if you want completely flat map. */ pitch: 30, duration: 650, /* Push vehicle lower on screen. This gives more visible map ahead of the vehicle. */ padding: { top: 80, bottom: 260, left: 30, right: 30 } }); /* ============================ SPEED / GPS DISPLAY ============================ */ let speedKmh = null; if (speed != null) { speedKmh = Math.max( 0, speed * 3.6 ); } document .getElementById("driveInfo") .innerHTML = ( speedKmh == null ? "" : `${speedKmh.toFixed(0)} km/h • ` ) + `GPS ±${Math.round(accuracy)} m`; /* Save current position so next GPS update can calculate heading. */ lastGps = { lat, lng }; }, error => { console.error( "Drive GPS error:", error ); status( "Please allow precise location." ); stopDrive(); }, { enableHighAccuracy: true, maximumAge: 0, timeout: 15000 } ); }
